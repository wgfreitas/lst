def resposta_ok(conteudo: str | None) -> httpx.Response:

def respostas_validas(n: int = EVENTOS) -> list[httpx.Response]:

def resposta_erro(
    status: int,
    mensagem: str,
    *,
    tipo: str = "invalid_request_error",
    param: str | None = None,
    code: str | None = None,
    headers: dict[str, str] | None = None,
) -> httpx.Response:
    corpo = {"error": {"message": mensagem, "type": tipo, "param": param, "code": code}}
    return httpx.Response(status, json=corpo, headers=headers)


def erro_response_format(tipo_rf: str = "json_schema") -> httpx.Response:
    return resposta_erro(
        400,
        f"Invalid parameter: 'response_format' of type '{tipo_rf}' is not supported with this model.",
        param="response_format",
    )


def erro_400_outro_motivo() -> httpx.Response:

def falha_transitoria(status: int = 503) -> httpx.Response:
    if status == 429:
        return resposta_erro(
            429,
            "Rate limit reached. Please try again later.",
            tipo="rate_limit_error",
            code="rate_limit_exceeded",
            headers=RETRY_RAPIDO,
        )
    return resposta_erro(
        status,
        "The server had an error while processing your request.",
        tipo="server_error",
        headers=RETRY_RAPIDO,
    )


def em_sequencia(*respostas: httpx.Response) -> Callable[[httpx.Request], httpx.Response]:
    """Devolve as respostas na ordem; uma requisição excedente recebe um 418 inequívoco."""
    fila = list(respostas)

    def responder(request: httpx.Request) -> httpx.Response:
        if fila:
            return fila.pop(0)
        return httpx.Response(
            418, json={"error": {"message": "Requisição excedente não prevista pelo teste."}}
        )

    return responder


def sempre(fabrica: Callable[[], httpx.Response]) -> Callable[[httpx.Request], httpx.Response]:
    return lambda request: fabrica()


def corpo_json(request: httpx.Request) -> dict[str, Any]:
    return json.loads(request.content.decode("utf-8"))


def corpos_enviados(rota: respx.Route) -> list[dict[str, Any]]:
    return [corpo_json(chamada.request) for chamada in rota.calls]


def tipo_response_format(corpo: dict[str, Any]) -> str | None:
    rf = corpo.get("response_format")
    return None if rf is None else rf.get("type")


def recusar_response_format(
    *recusados: str | None,
    fabrica_erro: Callable[[str], httpx.Response] | None = None,
) -> Callable[[httpx.Request], httpx.Response]:
    """Servidor que devolve 400 por response_format para os tipos recusados e 200 válido para os demais."""
    fabrica = fabrica_erro or erro_response_format
    contador = iter(range(10_000))

    def responder(request: httpx.Request) -> httpx.Response:
        tipo = tipo_response_format(corpo_json(request))
        if tipo in recusados:
            return fabrica(tipo)
        return resposta_valida(next(contador))

    return responder

def valor_simples(valor: Any) -> Any:
    valor = getattr(valor, "value", valor)
    if hasattr(valor, "get_secret_value"):
        valor = valor.get_secret_value()
    return valor


def assert_encerramento(resultado: Any, codigo: int) -> None:
    tudo = resultado.output + saida_erro(resultado)
    assert resultado.exit_code == codigo, tudo
    assert "Traceback" not in tudo
    if codigo == 0:
        assert resultado.exception is None, repr(resultado.exception)
    else:
        # Saída controlada (SystemExit); exceção não tratada também daria código 1 no CliRunner.
        assert isinstance(resultado.exception, SystemExit), repr(resultado.exception)
    assert CHAVE not in tudo

def test_um_post_por_evento_com_bearer_modelo_mensagens_e_max_tokens(servidor_llm, rota_chat, log_calibrado, criar_settings):
    """Cada evento marcado gera um POST em /chat/completions com Bearer, modelo padrão, system+user e max_tokens padrão."""
    rota_chat.mock(side_effect=em_sequencia(*respostas_validas()))
    executar_pipeline(log_calibrado, criar_settings())
    assert rota_chat.call_count == EVENTOS
    assert len(servidor_llm.calls) == EVENTOS
    for chamada in rota_chat.calls:
        req = chamada.request
        assert req.method == "POST"
        assert str(req.url) == CHAT_URL
        assert req.headers["Authorization"] == f"Bearer {CHAVE}"
        assert req.headers["Content-Type"].startswith("application/json")
        corpo = corpo_json(req)
        assert corpo["model"] == MODELO_PADRAO
        assert corpo["max_tokens"] == MAX_TOKENS_PADRAO
        assert [m["role"] for m in corpo["messages"]] == ["system", "user"]
        assert all(isinstance(m["content"], str) and m["content"].strip() for m in corpo["messages"])
    assert len({c["messages"][1]["content"] for c in corpos_enviados(rota_chat)}) == EVENTOS

def test_somente_template_e_amostras_limitadas_saem_da_maquina(rota_chat, log_calibrado, criar_settings):
    """Cada requisição leva no máximo 3 linhas originais do log e no máximo 3 IPs de amostra fora delas."""
    rota_chat.mock(side_effect=em_sequencia(*respostas_validas()))
    executar_pipeline(log_calibrado, criar_settings())
    linhas = {
        linha.strip()
        for linha in log_calibrado.read_text(encoding="utf-8", errors="replace").splitlines()
        if linha.strip()
    }
    for corpo in corpos_enviados(rota_chat):
        texto = "\n".join(m["content"] for m in corpo["messages"])
        presentes = {linha for linha in linhas if linha in texto}
        assert len(presentes) <= 3, presentes
        resto = corpo["messages"][1]["content"]
        for linha in sorted(presentes, key=len, reverse=True):
            resto = resto.replace(linha, "")
        assert len(set(IPV4.findall(resto))) <= 3
