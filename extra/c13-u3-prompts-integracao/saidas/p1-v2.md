def explicacao(n: int) -> str:
    return (f"Resposta simulada {n}: 14 falhas de login SSH para o usuário fictício suporte.lab "
            f"a partir de 203.0.113.{10 + n} em 90 segundos, padrão compatível com força bruta.")


def proxima_acao(n: int) -> str:
    return (f"Resposta simulada {n}: bloquear 203.0.113.{10 + n} no firewall de borda e verificar "
            "se houve login bem-sucedido de suporte.lab depois das falhas.")


def conteudo_valido(n: int = 1, severity: str = "high") -> str:
    return json.dumps(
        {"explanation": explicacao(n), "severity": severity, "next_action": proxima_acao(n)},
        ensure_ascii=False,
    )


def resposta_chat(conteudo: str | None) -> httpx.Response:
    return httpx.Response(200, json={
        "id": "chatcmpl-ficticio-0001",
        "object": "chat.completion",
        "created": 1767225600,
        "model": "gpt-oss:20b",
        "choices": [{"index": 0, "finish_reason": "stop",
                     "message": {"role": "assistant", "content": conteudo}}],
        "usage": {"prompt_tokens": 412, "completion_tokens": 96, "total_tokens": 508},
    })


def erro_api(status: int, mensagem: str, tipo: str = "invalid_request_error", param: str | None = None,
             code: str | None = None, headers: dict | None = None) -> httpx.Response:
    return httpx.Response(status, headers=headers,
                          json={"error": {"message": mensagem, "type": tipo, "param": param, "code": code}})


def falha_503() -> httpx.Response:
    return erro_api(503, "The server is overloaded. Please retry.", tipo="server_error")


def falha_401() -> httpx.Response:
    return erro_api(401, "Invalid API key provided.", tipo="authentication_error", code="invalid_api_key")


INVALIDO_NAO_JSON = "Não consegui concluir a análise deste evento."


def respostas(*primeiras):
    """side_effect: chama as fábricas de `primeiras` em ordem e, depois, devolve respostas válidas numeradas."""
    fila = list(primeiras)
    numero = itertools.count(1)

    def responder(request):
        if fila:
            return fila.pop(0)()
        return resposta_chat(conteudo_valido(next(numero)))

    return responder


def servidor_sem_response_format():
    """side_effect de um provedor que recusa qualquer response_format com 400 e responde bem sem ele."""
    numero = itertools.count(1)

    def responder(request):
        formato = json.loads(request.content).get("response_format")
        if formato is not None:
            return erro_api(400, f"Invalid parameter: 'response_format' of type '{formato.get('type')}' "
                                 "is not supported with this model.", param="response_format")
        return resposta_chat(conteudo_valido(next(numero)))

    return responder

def scan(runner, *opcoes, log: Path = LOG, env: dict | None = None):
    variaveis = {"LLM_API_KEY": CHAVE, **(env or {})}  # valor None remove a variável
    return runner.invoke(app, ["scan", *opcoes, str(log)], env=variaveis)


def pipeline(**campos) -> str:
    return asyncio.run(run_pipeline(LOG, Settings(_env_file=None, llm_api_key=CHAVE, **campos)))


def corpo(chamada) -> dict:
    return json.loads(chamada.request.content)


def tentativas_do_primeiro_evento(rota) -> int:
    """Conta as requisições idênticas à primeira (reenvios HTTP do mesmo evento)."""
    primeira = rota.calls[0].request.content
    return sum(1 for chamada in rota.calls if chamada.request.content == primeira)


def normalizar(texto: str) -> str:
    return " ".join(texto.split())


def assert_response_format(req: dict, modo: str) -> None:
    if modo == "json_schema":
        formato = req["response_format"]
        assert formato["type"] == "json_schema"
        assert formato["json_schema"]["name"] == "triage_explanation"
        assert formato["json_schema"]["strict"] is True
        assert set(formato["json_schema"]["schema"]["required"]) == CAMPOS
    elif modo == "json_object":
        assert req["response_format"] == {"type": "json_object"}
    else:
        assert "response_format" not in req


def assert_sem_excecao_interna(resultado) -> None:
    assert resultado.exception is None or isinstance(resultado.exception, SystemExit), (
        f"exceção interna escapou da CLI: {resultado.exception!r}")
    assert "Traceback (most recent call last)" not in resultado.stdout
    # Em sucesso, o stdout é o relatório (texto do LLM e do log); só o stderr é mensagem da CLI.
    texto = resultado.stderr if resultado.exit_code == 0 else resultado.stdout + resultado.stderr
    achado = EXCECAO_INTERNA.search(texto)
    assert achado is None, f"a saída expõe uma exceção interna: {achado.group(0)!r}"


def assert_requisitos_da_equipe(resultado, *causas: str) -> None:
    """Requisito da equipe: nenhuma exceção interna e, havendo erro, a mensagem nomeia a causa."""
    assert resultado.exit_code in (0, 1, 2)
    assert_sem_excecao_interna(resultado)
    if resultado.exit_code != 0:
        fluxo = resultado.stderr if resultado.exit_code == 1 else resultado.stdout + resultado.stderr
        mensagem = normalizar(fluxo).lower()
        assert mensagem, "erro sem mensagem para o analista"
        if causas:
            assert any(c.lower() in mensagem for c in causas), f"a mensagem não nomeia {causas}: {mensagem!r}"


def assert_relatorio(resultado, explicados: int) -> None:
    assert resultado.exit_code == 0, resultado.stderr
    assert_sem_excecao_interna(resultado)
    assert resultado.stdout.lstrip().startswith(CABECALHO)
    assert resultado.stdout.count("### Explicação") == explicados
    assert resultado.stdout.count("### Próxima ação") == explicados

def test_configuracao_padrao(runner, provedor):
    """[P01] Só com LLM_API_KEY: os padrões vão na requisição e o relatório traz os 3 eventos explicados."""
    rota = provedor.post(URL_PADRAO).mock(side_effect=respostas())
    resultado = scan(runner)
    assert_relatorio(resultado, explicados=EVENTOS)
    assert rota.call_count == EVENTOS
    for chamada in rota.calls:
        req = corpo(chamada)
        assert chamada.request.headers["Authorization"] == f"Bearer {CHAVE}"
        assert req["model"] == "gpt-oss:20b"
        assert req["max_tokens"] == 1024
        assert_response_format(req, "json_schema")
    saida = normalizar(resultado.stdout)
    for n in range(1, EVENTOS + 1):
        assert normalizar(explicacao(n)) in saida
        assert normalizar(proxima_acao(n)) in saida

def test_dados_enviados_ao_provedor(runner, provedor):

    assert resultado.exit_code == 0
    linhas_log = {linha.strip() for linha in LOG.read_text(encoding="utf-8", errors="replace").splitlines()
                  if len(linha.strip()) >= 20}
    assert rota.call_count == EVENTOS

        assert len({linha for linha in linhas_log if linha in usuario}) <= 3


def test_uma_chamada_por_evento_em_sequencia(runner, provedor):

    numero = itertools.count(1)

    async def responder(request):
        estado["ativas"] += 1
        estado["pico"] = max(estado["pico"], estado["ativas"])
        await asyncio.sleep(0.05)
        estado["ativas"] -= 1
        return resposta_chat(conteudo_valido(next(numero)))

    rota = provedor.post(URL_PADRAO).mock(side_effect=responder)

## 3. Lacunas do contrato

1. **`--dry-run`:** qual o código de saída e o que vai para o stdout? Gera relatório? Com quais seções?
2. **Tentativas esgotadas para 5xx/429:** o scan aborta com código 1 (com qual mensagem) ou o evento vai para "Eventos não explicados" e os demais continuam?
3. **400 em modo fixado ou por outro motivo:** qual o código de saída exato e o texto da mensagem? Sai um relatório parcial?
4. **Como reconhecer um "400 por causa de `response_format`":** pelo `error.param`, pelo texto da mensagem ou pelo `error.code`? E nos provedores que devolvem `{"error": "texto"}`, como o Ollama?
5. **Degradação no modo auto:** o formato degradado vale para os eventos seguintes ou cada evento recomeça em json_schema?
6. **403, 404, 3xx e demais 4xx (408, 409, 413, 422):** como são classificados (autenticação ou erro operacional)? Com que código de saída? São repetidos?
7. **Timeout e falha de conexão (DNS, conexão recusada, TLS):** contam como falha transitória para `LLM_MAX_RETRIES`? Qual código e qual mensagem para falha de conexão?
8. **Escopo do timeout:** vale por tentativa ou para a chamada inteira, com as novas tentativas? Cobre conexão e leitura? (P07 supõe que ele vira o timeout do cliente HTTP.)
9. **Espera entre tentativas:** existe backoff? O LST respeita `Retry-After`/`retry-after-ms`? Com que teto?
10. **200 com corpo fora do formato chat.completion** (HTML, `choices` vazio, `content: null`, objeto `error` com status 200, comum em agregadores): isso conta como "resposta inválida" (nova requisição e "não explicados") ou como erro operacional?
11. **JSON fora do esperado:** campos além dos três invalidam a resposta ("exatamente")? `explanation` ou `next_action` vazios são aceitos? Se houver vários objetos JSON no texto, qual vale? `finish_reason: "length"` muda algo?
12. **Severidade no relatório:** a severidade aparece? Em que formato? Espaços (" High ") são tolerados?
13. **Novas requisições de parse:** repetem o corpo ou acrescentam uma instrução corretiva? Um 5xx durante uma delas consome o orçamento de `LLM_MAX_RETRIES` ou o de `LLM_PARSE_RETRIES`?
14. **Seção "## Eventos não explicados":** como cada evento é identificado nela? Ela aparece quando está vazia? Qual a ordem dos eventos no relatório?
15. **Formato da mensagem `user` e escolha das amostras:** sem saber como as amostras são serializadas, não dá para verificar de fora o limite de 3 IPs e 3 usuários. P16 verifica só as linhas originais e a chave.
16. **Chave vazia:** `LLM_API_KEY` vazia ou só com espaços conta como ausente?
17. **Precedência:** se o ambiente e o `.env` definem a mesma variável, qual vale?
18. **Barra final no `LLM_BASE_URL`:** um valor como `…/v1/` é normalizado ou gera `//chat/completions`?
19. **Código 2:** a mensagem vai para o stderr? O código também vale para diretório no lugar de arquivo, arquivo sem permissão de leitura e log vazio? Um log sem eventos marcados gera relatório com código 0?
20. **Mascaramento da chave:** as mensagens de erro devem mascarar a chave, por exemplo quando o provedor devolve parte dela?
21. **Campos e cabeçalhos extras:** o corpo pode ter outros campos (`temperature`, `stream`, `seed`)? E cabeçalhos extras?
22. **Erro de validação de configuração:** qual o texto da mensagem? Ela lista todas as variáveis inválidas ou só a primeira?
23. **Exceções de `run_pipeline`:** que tipos ela propaga? Uma hierarquia própria do LST ou as exceções do SDK e do httpx?
